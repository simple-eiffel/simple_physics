Installation

simple_physics is a library for Eiffel 25.02+. Add it to your project using the ECF (Eiffel Configuration File).

Step 1: Add to Your ECF

<library name="simple_physics" location="$SIMPLE_EIFFEL/simple_physics/simple_physics.ecf"/>

Step 2: Set Compiler Settings

Ensure your project uses void-safe mode:

<capability>
    <void_safety support="all"/>
    <concurrency support="scoop"/>
</capability>

Step 3: Compile

ec -batch -config your_project.ecf -target your_target -c_compile

Getting Started

Your First Simulation

Create a simple falling object simulation:

class HELLO_PHYSICS
create
    make

feature {NONE} -- Initialization

    make
            -- Run a simple physics simulation.
        local
            l_physics: SIMPLE_PHYSICS
            l_world: PHYSICS_WORLD
            l_body: PHYSICS_BODY
            l_pos, l_vel, l_gravity: VECTOR_3D
            i: INTEGER
            l_dt: REAL_64
        do
            -- Create physics engine with gravity
            create l_physics.make
            l_physics.set_gravity (9.8).do_nothing
            l_world := l_physics.create_world

            -- Create a falling object at height 100 meters
            create l_pos.make (0.0, 100.0, 0.0)
            create l_vel.zero
            create l_body.make (10.0, 0.5, l_pos, l_vel)

            -- Add to world
            l_world.add_body (l_body)

            -- Simulate 10 seconds with 0.016 second time steps (60 FPS)
            l_dt := 0.016
            from i := 1 until i > (10.0 / l_dt).truncated_to_integer loop
                l_world.integrate_step (l_dt)
                print ("Time: " + ((i * l_dt).out) + "s, Height: " + l_body.position.y.out + "m%N")
                i := i + 1
            end

            print ("Impact at " + l_body.position.y.out + " meters%N")
        end

end

Understanding Vectors

Vectors represent positions, velocities, and forces in 3D space:

local
    l_pos, l_vel, l_result: VECTOR_3D
do
    -- Position of object at (1, 2, 3) meters
    create l_pos.make (1.0, 2.0, 3.0)

    -- Velocity of 5 m/s in x-direction
    create l_vel.make (5.0, 0.0, 0.0)

    -- Distance from origin
    print ("Distance: " + l_pos.magnitude.out)

    -- After 1 second, new position is old + velocity
    l_result := l_pos.add (l_vel)

    -- Direction of motion (unit vector)
    l_result := l_vel.normalize
end

Creating Bodies

Two types of bodies: dynamic (movable) and static (immovable):

local
    l_body_dynamic, l_body_static: PHYSICS_BODY
    l_pos, l_vel: VECTOR_3D
do
    -- Dynamic body: falls under gravity, affected by forces
    create l_pos.make (0.0, 50.0, 0.0)
    create l_vel.zero
    create l_body_dynamic.make (
        5.0,      -- mass in kg
        0.5,      -- radius in meters
        l_pos,    -- initial position
        l_vel     -- initial velocity
    )

    -- Static body: immovable, like a floor
    create l_pos.make (0.0, 0.0, 0.0)
    create l_body_static.make_static (
        1000.0,   -- mass (not used for statics, but required)
        l_pos     -- position
    )

    -- Verify types
    if l_body_dynamic.body_type.is_dynamic then
        print ("Body is dynamic%N")
    end

    if l_body_static.body_type.is_static then
        print ("Body is static%N")
    end
end

Common Patterns

Simulation Loop

The standard pattern for running a physics simulation:

-- Setup
l_world.set_gravity (create {VECTOR_3D}.make (0.0, -9.8, 0.0))

-- Main loop
from i := 1 until i > max_frames loop
    -- Advance simulation by 0.016 seconds (60 FPS)
    l_result := l_world.integrate_step (0.016)

    -- Process each body
    across l_world.bodies as l_body_cursor loop
        -- Update display, check collisions, etc.
        process_body (l_body_cursor.item)
    end

    -- Increment frame
    i := i + 1
end

Energy Calculations

Monitor total energy in the system (kinetic + potential):

local
    l_total_ke, l_total_pe, l_total_energy: REAL_64
    l_gravity: REAL_64
do
    l_gravity := 9.8
    l_total_ke := 0.0
    l_total_pe := 0.0

    across l_world.bodies as l_body_cursor loop
        if l_body_cursor.item.body_type.is_dynamic then
            l_total_ke := l_total_ke + l_body_cursor.item.kinetic_energy
        end
        l_total_pe := l_total_pe + l_body_cursor.item.potential_energy (l_gravity)
    end

    l_total_energy := l_total_ke + l_total_pe
    print ("KE: " + l_total_ke.out + " J, PE: " + l_total_pe.out + " J, Total: " + l_total_energy.out + " J%N")
end

Checking Body Type

Handle dynamic and static bodies differently:

across l_world.bodies as l_body_cursor loop
    if l_body_cursor.item.body_type.is_dynamic then
        -- Compute physics for moving object
        apply_gravity (l_body_cursor.item)
        resolve_collisions (l_body_cursor.item)
    else
        -- Static body: no updates needed
        -- (still participates in collision detection)
    end
end

Modifying Bodies at Runtime

Change position and velocity during simulation:

local
    l_new_pos, l_new_vel: VECTOR_3D
do
    -- Teleport object
    create l_new_pos.make (10.0, 50.0, 0.0)
    l_body.set_position (l_new_pos)

    -- Change velocity
    create l_new_vel.make (5.0, 0.0, 0.0)
    l_body.set_velocity (l_new_vel)
end

Best Practices

1. Immutable Vectors

Vectors are immutable. Operations return new vectors:

-- ✓ Correct: Create new vector
l_result := l_v1.add (l_v2)

-- ✗ Wrong: Can't modify in place
l_v1.x := 10.0  -- Error: x is read-only

2. Time Step Consistency

Use fixed time steps for physics stability:

-- ✓ Correct: Fixed 60 FPS
l_world.integrate_step (0.016)

-- ✗ Wrong: Variable time steps cause instability
l_world.integrate_step (dt)  -- dt varies frame-to-frame

3. Gravity Direction

Gravity is a vector. Negative Y is down (Earth-like):

-- ✓ Correct: Gravity points downward
create l_gravity.make (0.0, -9.8, 0.0)
l_world.set_gravity (l_gravity)

-- ✗ Wrong: Positive Y makes objects fall up
create l_gravity.make (0.0, 9.8, 0.0)

4. Fluent Configuration

Set up the world before adding bodies:

-- ✓ Correct: Configure first
l_physics.set_gravity (9.8).do_nothing
l_world := l_physics.create_world
l_world.add_body (l_body)

-- ✗ Wrong: Configuring after bodies may be inconsistent
l_world.add_body (l_body)
l_physics.set_gravity (9.8).do_nothing

5. Memory Management

Bodies are managed by the world. Remove when done:

-- Add body to world
l_world.add_body (l_body)

-- ... later ...

-- Remove when no longer needed
l_world.remove_body (l_body)

Performance Tips

Body Count

Collision detection is O(n²). Keep body count reasonable:

  • 0-50 bodies: No performance issues
  • 50-200 bodies: Monitor performance
  • 200+ bodies: Consider spatial partitioning (not yet in simple_physics)

Time Step Size

Smaller steps = more accurate but slower:

  • 0.016s (60 FPS): Good for real-time visualization
  • 0.008s (120 FPS): More accurate, 2x slower
  • 0.033s (30 FPS): Less accurate, faster (not recommended)

Sphere Radius

Smaller radius = more collision checks:

  • Keep radii consistent with body count
  • Avoid very small radii (< 0.01m) unless necessary
  • Static bodies don't need accurate radii for collision

Troubleshooting

Objects Fall Too Slowly

Problem: Gravity seems weak.

Solution: Check gravity vector direction and magnitude:

-- Gravity should be negative Y for downward
create l_gravity.make (0.0, -9.8, 0.0)  -- Correct
create l_gravity.make (0.0, 9.8, 0.0)   -- Wrong!

Objects Jump or Teleport

Problem: Bodies move erratically or far between frames.

Solution: Use consistent time steps and reasonable velocities:

-- ✓ Correct: Reasonable velocity
create l_vel.make (5.0, 0.0, 0.0)  -- 5 m/s

-- ✗ Wrong: Extremely high velocity
create l_vel.make (5000.0, 0.0, 0.0)  -- 5000 m/s!

Energy Increases Over Time

Problem: Total energy (KE + PE) grows without input.

Solution: This can happen with large time steps. Try smaller steps:

-- Try 0.008 instead of 0.016
l_world.integrate_step (0.008)

Collisions Don't Seem Right

Problem: Bodies pass through each other.

Solution: Ensure body radii are set appropriately:

-- Bodies should have realistic radii
create l_body.make (
    10.0,    -- mass
    0.5,     -- radius (500 mm - not too small!)
    l_pos,
    l_vel
)

Advanced Topics

RK4 Integration

simple_physics uses fourth-order Runge-Kutta (RK4) integration for accurate physics simulation. This means forces are evaluated at four points within each time step for better accuracy than simpler methods.

Sphere Collision Detection

Collision detection uses sphere-sphere algorithms. Two spheres collide when the distance between centers is less than the sum of their radii.

Impulse-Based Resolution

When bodies collide, simple_physics applies impulses (instantaneous changes in velocity) to simulate realistic bouncing.

Design by Contract

All features are protected by Design by Contract (DBC) specifications:

  • Preconditions: What must be true before calling a feature
  • Postconditions: What will be true after calling a feature
  • Invariants: What must always be true for an object

These contracts serve as executable documentation and catch bugs at runtime.
